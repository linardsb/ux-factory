// What actually costs during a ctrl-wheel zoom? Two candidates: arrow redraws, or re-rasterising
// 30 scaled compositions. Counted, not reasoned about.
import { createRequire } from "node:module";
import path from "node:path";
const MAIN = "/Users/Berzins/Desktop/Linards_current/ux-factory";
const require = createRequire(path.join(MAIN, "tooling", "visual-regression") + path.sep);
const pw = require("@playwright/test");
const b = await pw.chromium.launch();
const ctx = await b.newContext({ viewport: { width: 1440, height: 1000 } });
const p = await ctx.newPage();
await p.goto("http://127.0.0.1:4759/.claude/plans/canvas-spike-s1/harness.html?cfg=a&frames=30", { waitUntil: "load" });
await p.waitForSelector("html[data-ready='1']");
await p.waitForTimeout(500);
const cdp = await ctx.newCDPSession(p);
await cdp.send("Emulation.setCPUThrottlingRate", { rate: 4 });
const before = await p.evaluate(() => ({ redraws: window.__arrowRedraws, scrollend: window.__scrollendSeen }));
let wheels = 0;
await p.mouse.move(700, 400); await p.keyboard.down("Control");
for (let i = 0; i < 36; i++) { await p.mouse.wheel(0, 60); wheels++; await p.waitForTimeout(15); }
for (let i = 0; i < 36; i++) { await p.mouse.wheel(0, -60); wheels++; await p.waitForTimeout(15); }
await p.keyboard.up("Control");
await p.waitForTimeout(300);
const after = await p.evaluate(() => ({ redraws: window.__arrowRedraws, scrollend: window.__scrollendSeen,
  scaleWrites: window.__sxScale }));
await cdp.send("Emulation.setCPUThrottlingRate", { rate: 1 });
console.log(`wheel events dispatched : ${wheels}`);
console.log(`arrow redraws during zoom: ${after.redraws - before.redraws}   <- if 0, arrows are NOT the zoom cost`);
console.log(`scrollend during zoom    : ${after.scrollend - before.scrollend}`);
await b.close();
