// The rule, as the plan specifies it: BFS from the entry frame, one column per rank.
// connections are [affordanceId, placeId]; an affordance belongs to a place.
function rankLayout(board) {
  const places = board.places.map((p) => p.id);
  if (!places.length) return [];
  const ownerOf = new Map();
  for (const p of board.places) for (const a of p.affordances) ownerOf.set(a.id, p.id);
  const out = new Map(); // placeId -> { rank, order }
  const adj = new Map(places.map((id) => [id, []]));
  for (const [affId, toId] of board.connections || []) {
    const from = ownerOf.get(affId);
    if (from && adj.has(from) && places.includes(toId)) adj.get(from).push(toId);
  }
  const entry = places[0];                       // board order: the entry place is first
  const seen = new Set([entry]);                 // THE CYCLE GUARD
  let frontier = [entry], rank = 0;
  while (frontier.length) {
    frontier.forEach((id, i) => out.set(id, { rank, order: i }));
    const next = [];
    for (const id of frontier) for (const to of adj.get(id) || [])
      if (!seen.has(to)) { seen.add(to); next.push(to); }
    frontier = next; rank += 1;
  }
  // unreachable places get their own trailing rank, in board order — totality
  const orphans = places.filter((id) => !out.has(id));
  orphans.forEach((id, i) => out.set(id, { rank, order: i }));
  return places.map((id) => ({ id, ...out.get(id) }));
}

}
