// P1 — prove a FUNCTION-SCOPED group-7 predicate can be written, and that it reddens on all
// three mutations a file-scoped allowlist would miss. Run over a synthetic studio-canvas.mjs
// that carries the two named writers, plus the REAL MODULES list from build-checks.
import { readFileSync } from "node:fs";
const R = "/Users/Berzins/Desktop/Linards_current/ux-factory";

const STYLE_WRITE    = /\.setProperty\(|\.style\.[A-Za-z]\w*\s*=[^=]/g;
const ALLOWED_DIRECT = /\.style\.viewTransitionName\s*=[^=]/g;

// Slice a function body by brace-matching from its declaration. Written as the gate would have
// to write it — no parser, no dependency.
function sliceFn(src, name) {
  const decl = new RegExp(`(?:function\\s+${name}\\s*\\(|(?:const|let)\\s+${name}\\s*=\\s*(?:\\([^)]*\\)|[A-Za-z_$][\\w$]*)\\s*=>\\s*\\{)`);
  const m = decl.exec(src);
  if (!m) return null;
  let i = src.indexOf("{", m.index + m[0].length - 1);
  if (i < 0) return null;
  let depth = 0;
  for (let j = i; j < src.length; j += 1) {
    if (src[j] === "{") depth += 1;
    else if (src[j] === "}") { depth -= 1; if (!depth) return src.slice(i, j + 1); }
  }
  return null;
}

const count = (s) => ((s || "").match(STYLE_WRITE) || []).length - ((s || "").match(ALLOWED_DIRECT) || []).length;

// The predicate: per file, either a file-scoped exception (build-import.mjs) or a NAMED-WRITER
// budget; and the file's total must equal the sum of its slices, so a write anywhere else reddens.
const NAMED = { "studio-canvas.mjs": { setPos: 3, setScale: 3 } };
function checkFile(file, src) {
  const fails = [];
  const total = count(src);
  if (file === "build-import.mjs") {
    if (total !== 1) fails.push(`${file}: ${total} inline-style writes; applyToStage is the only one`);
    return fails;
  }
  const named = NAMED[file];
  if (!named) {
    if (total) fails.push(`${file} writes an inline style; only applyToStage, setPos and setScale may`);
    return fails;
  }
  let inSlices = 0;
  for (const [fn, want] of Object.entries(named)) {
    const body = sliceFn(src, fn);
    if (body === null) { fails.push(`${file}: named writer ${fn}() not found — the slice cannot be taken`); continue; }
    const got = count(body);
    inSlices += got;
    if (got !== want) fails.push(`${file}: ${fn}() makes ${got} inline-style writes, the invariant is ${want}`);
  }
  if (total !== inSlices) fails.push(`${file}: ${total - inSlices} inline-style write(s) OUTSIDE ${Object.keys(named).join(" / ")}`);
  return fails;
}

// A synthetic post-swap studio-canvas.mjs: the two named writers, 3 setProperty calls each.
const GOOD = `
export function setPos(el, x, y, w) {
  el.style.setProperty("--x", \`\${x}px\`);
  el.style.setProperty("--y", \`\${y}px\`);
  el.style.setProperty("--w", \`\${w}px\`);
}
export function setScale(stage, s) {
  stage.style.setProperty("--stx-scale", String(s));
  stage.style.setProperty("--stx-extent-w", \`\${1000 * s}px\`);
  stage.style.setProperty("--stx-extent-h", \`\${800 * s}px\`);
}
export function place(node) { node.setAttribute("data-stx-name", "x"); return node; }
`;
const real = (f) => readFileSync(`${R}/system/${f}`, "utf8");

const CASES = [
  ["CONTROL · the good tree",                  { "studio-canvas.mjs": GOOD }],
  ["M1 · a write OUTSIDE the named writers",   { "studio-canvas.mjs": GOOD.replace('export function place(node) {', 'export function place(node) {\n  node.style.opacity = "1";') }],
  ["M2 · a write in a module with no writer",  { "studio-verbs.mjs": real("studio-verbs.mjs") + '\nfunction leak(n){ n.style.setProperty("--x","0px"); }\n' }],
  ["M3 · a FOURTH write inside setPos",        { "studio-canvas.mjs": GOOD.replace('el.style.setProperty("--w"', 'el.style.setProperty("--z", "0");\n  el.style.setProperty("--w"') }],
  ["M4 · the file-scoped shortcut (the trap)", { "studio-canvas.mjs": GOOD.replace('export function place(node) {', 'export function place(node) {\n  node.style.left = "0px";\n  node.style.top = "0px";') }],
];
for (const [label, over] of CASES) {
  const fails = [];
  for (const f of ["studio-canvas.mjs", "studio-verbs.mjs", "studio.mjs", "build-import.mjs", "breadboard.mjs"])
    fails.push(...checkFile(f, over[f] ?? real(f)));
  console.log(`${fails.length ? "RED  " : "green"}  ${label}`);
  for (const f of fails) console.log(`         · ${f}`);
}
