import { readFileSync, existsSync, rmSync, cpSync } from "node:fs";
import pw from "/Users/Berzins/node_modules/playwright/index.js";
const S = process.env.S, W = `${S}/w`, P = readFileSync(`${S}/port`, "utf8").trim(), base = `http://localhost:${P}`;
const { renderPack, readBuildPackage } = await import(`${W}/agent-layer/gen-build-handoff.mjs`);
const pkg = `${S}/jobs/_discovery/fp-lanes`;
const ledger = () => readFileSync(`${pkg}/build/ops.jsonl`, "utf8").trim().split("\n").map(JSON.parse);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function waitLines(n) { for (let i = 0; i < 60; i++) { if (ledger().length >= n) break; await sleep(100); } return ledger(); }
let fails = 0; const t = (n, c, x = "") => { console.log(`${c ? "✓" : "✗"} ${n} ${c ? "" : x}`); if (!c) fails++; };
for (const engine of (process.argv[2] ?? "chromium,firefox,webkit").split(",")) {
  console.log(`— ${engine}`);
  rmSync(pkg + "/build", { recursive: true, force: true }); cpSync(`${W}/discovery/faster-payment/build`, `${pkg}/build`, { recursive: true }); rmSync(`${pkg}/build/handoff`, { recursive: true, force: true });
  const browser = await pw[engine].launch(); const page = await browser.newPage({ viewport: { width: 1440, height: 1000 } });
  const errs = []; page.on("pageerror", (e) => errs.push(e.message)); page.on("console", (m) => m.type() === "error" && errs.push(m.text()));
  await page.goto(`${base}/canvas.html?provenance=real&slug=fp-lanes`); await page.waitForSelector('html[data-canvas-page="ready"]');
  const node = (id) => page.locator(`[data-stx-id="${id}"]`);
  const flow = () => page.locator("[data-canvas-flow-text]").textContent();
  t("L1 select holds A only", (await page.locator("[data-canvas-lane] option").allTextContents()).join("|") === "A (base)");
  t("L1 keep hidden", await page.locator("[data-canvas-verb=lane-keep]").evaluate((b) => getComputedStyle(b).display === "none"));
  await page.click("[data-canvas-verb=lane-new]"); await page.fill("[data-canvas-lane-key]", "a"); await page.click("[data-canvas-verb=lane-create]");
  t("L2 key a refused, nothing drafted", (await page.evaluate(() => import("/canvas.mjs").then((m) => m.getCanvasPage().laneDraft))) === null);
  await page.fill("[data-canvas-lane-key]", "b"); await page.click("[data-canvas-verb=lane-create]");
  t("L2 draft b", (await page.locator("[data-canvas-lane]").inputValue()) === "b" && ledger().length === 6);
  await page.click('[data-cv-details="f1"]'); await page.selectOption("#cv-lane-part", "continue"); await page.selectOption("#cv-lane-prop", "label"); await page.fill("#cv-lane-value", "Check the name");
  await page.getByRole("button", { name: "Set in lane" }).click();
  t("L3 f1 relabelled in b, nothing written", (await node("f1").textContent()).includes("Check the name") && ledger().length === 6);
  await page.click('[data-cv-details="f2"]'); await page.getByRole("button", { name: "Leave out of lane" }).click();
  t("L4 f2 not in lane b", (await node("f2").textContent()).includes("Not in lane b"));
  t("L4 error chip is not an ask", (await node("f1").textContent()).includes("error: missing in lane b") && (await page.locator('[data-cv-ask-state="f1:error"]').count()) === 0);
  t("L4 flow has no f2", !(await flow()).includes("f2"));
  await page.click("[data-canvas-verb=lane-keep]");
  const L = await waitLines(7);
  t("L5 one variant.add line", JSON.stringify(L[6]?.params) === JSON.stringify({ key: "b", overrides: { f1: { set: { continue: { label: "Check the name" } } }, f2: { omit: true } } }) && L[6].status === "applied" && L[6].source === "owner", JSON.stringify(L[6]));
  for (let i = 0; i < 40 && !existsSync(`${pkg}/build/handoff/flow.md`); i++) await sleep(100);
  t("L5 save regenerated the pack", existsSync(`${pkg}/build/handoff/flow.md`) && readFileSync(`${pkg}/build/handoff/flow.md`, "utf8").includes("## Lane b"));
  await page.selectOption("[data-canvas-lane]", "");
  t("L6 A: Continue, asks without error, flow has f1 --> f2", (await node("f1").textContent()).includes("Continue") && (await page.locator('[data-cv-ask-state="f1:error"]').count()) === 0 && (await page.locator('[data-cv-ask-state="f1:empty"]').count()) === 1 && (await flow()).includes("f1 --> f2"));
  await page.selectOption("[data-canvas-lane]", "b");
  t("L6 b: no f1 --> f2, missing names error", !(await flow()).includes("f1 --> f2") && (await page.locator("[data-canvas-flow-missing]").textContent()).includes("error") && ledger().length === 7);
  await page.click("[data-canvas-verb=pack]");
  await page.waitForFunction(() => document.querySelector("[data-canvas-save]").textContent.startsWith("Handoff pack written"), null, { timeout: 5000 }).catch(() => {});
  t("L7 pack equals renderPack", readFileSync(`${pkg}/build/handoff/flow.md`, "utf8") === renderPack(readBuildPackage(pkg))["flow.md"]);

  await page.locator(".stx-scroll").focus();
  await page.keyboard.press("ControlOrMeta+z");
  const U = await waitLines(8);
  t("L8 undo restates variant.add", U[7]?.status === "undone" && U[7]?.op === "variant.add", JSON.stringify(U[7]));
  t("L8 select back to A only", (await page.locator("[data-canvas-lane] option").allTextContents()).join("|") === "A (base)");
  t("L9 no page errors", errs.length === 0, errs.join(" | "));
  await browser.close();
}
console.log(fails ? `✗ ${fails} failed` : "✓ all passed"); process.exit(fails ? 1 : 0);
