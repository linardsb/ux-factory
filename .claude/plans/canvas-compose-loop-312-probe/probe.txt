// vocabContext — driver.txt:69-87 verbatim
function vocabContext(vocab, vocabSha = VOCAB_SHA) {
  const firstSentence = (u = '') => { const cut = u.search(/\.\s|\n/); return (cut >= 0 ? u.slice(0, cut + 1) : u).trim().slice(0, 160); };
  const blocks = Object.entries(vocab.components).map(([name, c]) => {
    const props = Object.entries(c.props).map(([p, s]) => `  ${p}${s.required ? '!' : ''}: ${s.enum ? `enum ${s.enum.join('|')}` : s.type}`);
    const kids = c.children.length ? `${c.children.join(', ')}${c.childrenCardinality === 'many' ? ' (many)' : ' (one)'}` : 'none';
    return [`### ${name}`, ...props, `  children: ${kids}`, `  ${firstSentence(c.usage)}`].join('\n');
  });
  return [`Generated from ${vocab.generatedFrom} (vocabulary.json sha256 ${vocabSha}). \`!\` marks a required prop.`, `Shape: ${vocab.composition.shape}`, `Children: ${vocab.composition.childrenRule}`, '', ...blocks].join('\n');
}

function idProblem(tree, p = 'composition') {
  const seen = new Set();
  const walk = (n, at) => {
    if (!n || typeof n !== 'object') return `${at} is not a node`;
    const isRoot = at === p;
    if (!isRoot && (typeof n.id !== 'string' || !n.id.trim())) return `${at} (${n.name}) has no id`;
    if (typeof n.id === 'string') { if (seen.has(n.id)) return `${at} (${n.name}) repeats id "${n.id}"`; seen.add(n.id); }
    for (const [i, c] of (n.children ?? []).entries()) { const e = walk(c, `${at}.children[${i}]`); if (e) return e; }
    return null;
  };
  return walk(tree, p);
}

const state = { doc: emptyDoc(), ask: null, calls: 0, write: null };
function handle(tool, args) {
  state.calls += 1;
  const refuse = (kind, error) => { state.write({ k: 'refused', tool, kind, error, args }); return { isError: true, content: [{ type: 'text', text: `refused: ${error}` }] }; };
  if (state.calls > 1) return refuse('one-per-turn', 'this turn already made its one proposal call — the owner decides before the next turn (LOOP)');
  const want = state.ask.kind === 'screen' ? 'screen_compose' : 'state_add';
  if (tool !== want) return refuse('wrong-target', `this turn asks for ${want}`);
  try {
    if (tool === 'screen_compose') {
      const params = Object.fromEntries(Object.entries({ screenId: args.screenId, why: args.why, composition: args.composition, decisionRefs: args.decisionRefs }).filter(([, v]) => v !== undefined));
      try { validateComposition(VOCAB, params.composition); } catch (e) { return refuse('vocabulary', e.message); }
      const idp = idProblem(params.composition); if (idp) return refuse('ids', `${idp} — every part needs an id so its states can address it`);
      let next; try { next = applyOp(state.doc, { op: 'screen.compose', params }); } catch (e) { return refuse('applier', e.message); }
      state.doc = next; const fid = next.frames.at(-1).id;
      state.write({ k: 'op', status: 'proposed', op: 'screen.compose', frameId: fid, args });
      return { content: [{ type: 'text', text: `filed ${fid}: screen.compose "${args.screenId}" (proposed — the owner decides)` }] };
    }
    if (args.baseId !== state.ask.baseId || args.stateKey !== state.ask.stateKey) return refuse('wrong-target', `this turn asks for the ${state.ask.stateKey} state of ${state.ask.baseId}`);
    if (args.override?.add !== undefined) return refuse('applier', 'override.add is not rendered yet (G19)');
    if (typeof args.why !== 'string' || !args.why.trim()) return refuse('applier', 'why must name the decision (D4)');
    let next; try { next = applyOp(state.doc, { op: 'state.add', params: { baseId: args.baseId, stateKey: args.stateKey, override: args.override } }); } catch (e) { return refuse('applier', e.message); }
    const fid = next.frames.at(-1).id; const t = frameTree(next, fid);
    const dangling = t.flags.find((f) => /^dangling/.test(f.kind)); if (dangling) return refuse('applier', `override names part "${dangling.partId}", which the base does not have`);
    try { validateComposition(VOCAB, t.tree); } catch (e) { return refuse('vocabulary', e.message); }
    state.doc = next;
    state.write({ k: 'op', status: 'proposed', op: 'state.add', frameId: fid, args });
    return { content: [{ type: 'text', text: `filed ${fid}: state.add "${args.stateKey} of ${args.baseId}" (proposed — the owner decides)` }] };
  } catch (e) { return refuse('bug', e.message); }
}
