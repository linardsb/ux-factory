// C1 — the CIEDE2000 port against skimage.color.deltaE_ciede2000, the SAME function E4 calls.
import { execFileSync } from "node:child_process";
import { ciede2000, rgbToLab, deltaEHex } from "./compare.mjs";

// A fixed-seed LCG so the 500 pairs are the same on every run and in the mutated half.
let seed = 20260917;
const rnd = () => (seed = (seed * 1103515245 + 12345) & 0x7fffffff) / 0x7fffffff;
const pairs = [];
for (let i = 0; i < 500; i++) {
  pairs.push([
    [Math.floor(rnd() * 256), Math.floor(rnd() * 256), Math.floor(rnd() * 256)],
    [Math.floor(rnd() * 256), Math.floor(rnd() * 256), Math.floor(rnd() * 256)],
  ]);
}
const mine = pairs.map(([a, b]) => ciede2000(rgbToLab(a), rgbToLab(b)));

const PY = process.env.HOME + "/Desktop/email-hub/.venv/bin/python";
const script = `
import sys, json, numpy as np
from skimage.color import rgb2lab, deltaE_ciede2000
d = json.load(sys.stdin)
a = np.array([p[0] for p in d], dtype=float).reshape(-1,1,3) / 255.0
b = np.array([p[1] for p in d], dtype=float).reshape(-1,1,3) / 255.0
out = deltaE_ciede2000(rgb2lab(a), rgb2lab(b)).reshape(-1)
print(json.dumps(out.tolist()))
`;
const theirs = JSON.parse(execFileSync(PY, ["-c", script], { input: JSON.stringify(pairs), encoding: "utf8", maxBuffer: 1 << 24 }));
let maxAbs = 0, where = -1;
for (let i = 0; i < mine.length; i++) {
  const d = Math.abs(mine[i] - theirs[i]);
  if (d > maxAbs) { maxAbs = d; where = i; }
}
console.log(`C1  oracle=${PY} (skimage), ${pairs.length} pseudo-random sRGB pairs, fixed seed ${20260917}`);
console.log(`C1  max |port - skimage| = ${maxAbs.toExponential(4)}  at pair ${where} ${JSON.stringify(pairs[where])}`);
console.log(`C1  port=${mine[where].toFixed(6)}  skimage=${theirs[where].toFixed(6)}`);
// the four named pairs from the plan's Task 7 VALIDATE
for (const [h1, h2, want] of [["#454545","#0c3b2f",18.2154],["#575757","#20828d",23.1670],["#f8f8f8","#ffffff",1.4004],["#000000","#ffffff",100.0]]) {
  const got = deltaEHex(h1, h2);
  console.log(`C1  ${h1}/${h2} = ${got.toFixed(4)}  (plan: ${want.toFixed(4)}, |diff| ${Math.abs(got-want).toExponential(2)})`);
}
console.log(maxAbs < 1e-3 ? "C1 PASS  max abs diff < 1e-3" : `C1 FAIL  max abs diff ${maxAbs} >= 1e-3`);
process.exit(maxAbs < 1e-3 ? 0 : 1);
