// S3 — checkPairs over BOTH packs (ticket #300). PARKED as wcag-probe.txt.
//   cp wcag-probe.txt /tmp/wcag-probe.mjs && node /tmp/wcag-probe.mjs
// The point of this probe is the half that PASSES: run 3's pack is 12/12 green while its body text is
// a dark green. A formal contrast check cannot distinguish a right colour from a wrong one.
import { readFileSync } from "node:fs";
import path from "node:path";
const MAIN = "/Users/Berzins/Desktop/Linards_current/ux-factory";
const SPIKE = process.env.SPIKE_DIR || path.join(process.env.HOME, "Desktop/Linards_current/wt-s3-300/.claude/plans/canvas-spike-s3");
const { checkPairs } = await import(path.join(MAIN, "system", "wcag.mjs"));
const { RULESET } = await import(path.join(MAIN, "system", "derive.rules.mjs"));

// checkPairs THROWS if any pair token is not #rrggbb (wcag.mjs:30). The faithful pack maps two roles
// (four under M2), so it must be OVERLAID on a full token map — a two-token object alone throws on
// pair 5. tokens.neutral.css is that full map.
// tokens.neutral.css binds the SEMANTIC tokens to PRIMITIVES through var() (`--color-fg:
// var(--color-ink)`), so a hex-only regex returns the primitives and none of the roles checkPairs
// asks for. Resolve the alias chain, then keep only what resolved to a hex.
const parse = (css) => {
  const raw = Object.fromEntries([...css.matchAll(/^\s*--([a-z0-9-]+):\s*([^;]+);/gm)].map((m) => [m[1], m[2].trim()]));
  const out = {};
  for (const k of Object.keys(raw)) {
    let v = raw[k];
    for (let hops = 0; hops < 8 && /^var\(\s*--([a-z0-9-]+)/.test(v); hops++) {
      v = raw[v.match(/^var\(\s*--([a-z0-9-]+)/)[1]] ?? "";
    }
    if (/^#[0-9a-fA-F]{6}$/.test(v)) out[k] = v.toLowerCase();
  }
  return out;
};
const neutral = parse(readFileSync(path.join(MAIN, "system", "tokens.neutral.css"), "utf8"));
const mapping = JSON.parse(readFileSync(path.join(SPIKE, "mapping.json"), "utf8"));
const strip = (o) => Object.fromEntries(Object.entries(o).map(([k, v]) => [k.replace(/^--/, ""), v]));

const PACKS = [
  ["neutral (the contract's own pack, for reference)", neutral],
  ["faithful (the source's own literals on the mapped roles, over neutral)", { ...neutral, ...strip(mapping.packs.faithful.values) }],
  ["polaris (run 3) — tokens.polaris.spike.css, the REAL file", { ...neutral, ...parse(readFileSync(path.join(MAIN, ".claude/plans/design-import-spike-a/tokens.polaris.spike.css"), "utf8")) }],
];
for (const [label, tokens] of PACKS) {
  console.log(`\n=== ${label} ===`);
  const rows = checkPairs(tokens, RULESET.wcagPairs);
  for (const r of rows) {
    console.log(`  ${r.pass ? "pass" : "FAIL"}  ${String(r.ratio).padStart(6)}:1  (min ${r.min})  ${r.fg} ${tokens[r.fg]} on ${r.bg} ${tokens[r.bg]}  — ${r.usage}`);
  }
  console.log(`  ${label.split(" ")[0]}: ${rows.filter((r) => r.pass).length}/${rows.length} pass`);
}
console.log(`\npolaris (run 3): ${checkPairs(PACKS[2][1], RULESET.wcagPairs).filter((r) => r.pass).length}/12 pass`);
console.log(`--color-fg is ${PACKS[2][1]["color-fg"]} — Polaris's color-text-success-strong, a dark green, on the BODY TEXT role.`);
console.log(`Every declared pair is green. The formal check has no dimension that can say so.`);
