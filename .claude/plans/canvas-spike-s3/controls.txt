// Half 1 runs every control against the PRISTINE metric. Half 2 applies each control's mutation and
// re-runs only that control, expecting its own named failure. A mutation that does not change the

//   * INPUT mutations feed the pristine metric a deliberately wrong input — they break the harness.
// C1/C2/C5/C9/C10 are the first kind; C3/C4b/C7/C8 the second. C6 has no mutation by design.

// ------------------------------------------------------------------ helpers
const clone = (img) => ({ w: img.w, h: img.h, data: Buffer.from(img.data) });
const setPx = (img, x, y, [r, g, b]) => { const i = (y * img.w + x) * 3; img.data[i] = r; img.data[i + 1] = g; img.data[i + 2] = b; };
const getPx = (img, x, y) => { const i = (y * img.w + x) * 3; return [img.data[i], img.data[i + 1], img.data[i + 2]]; };
const hexOf = ([r, g, b]) => "#" + [r, g, b].map((v) => v.toString(16).padStart(2, "0")).join("");
const HEX = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16));
const blank = (w, h, c) => { const im = { w, h, data: Buffer.alloc(w * h * 3) }; for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) setPx(im, x, y, c); return im; };
const shift = (img, dx, dy) => {           // translate an image, clamping at the edges
  const out = clone(img);
  for (let y = 0; y < img.h; y++) for (let x = 0; x < img.w; x++) {
    const sx = Math.min(img.w - 1, Math.max(0, x - dx)), sy = Math.min(img.h - 1, Math.max(0, y - dy));
    setPx(out, x, y, getPx(img, sx, sy));
  }
  return out;
};
const statsOne = (M, A, B, r) => M.regionStats(A, B, M.blur(A, 1.5), M.blur(B, 1.5), r);
const part = (M, name) => M.PARTS().find((r) => r.name === name);

// C1 — the CIEDE2000 port equals skimage.color.deltaE_ciede2000 over 500 pseudo-random pairs.
function C1(M, half) {
  let seed = 20260917;
  const rnd = () => (seed = (seed * 1103515245 + 12345) & 0x7fffffff) / 0x7fffffff;
  const pairs = [];
  for (let i = 0; i < 500; i++) pairs.push([[0,0,0].map(() => Math.floor(rnd() * 256)), [0,0,0].map(() => Math.floor(rnd() * 256))]);
  const mine = pairs.map(([a, b]) => M.ciede2000(M.rgbToLab(a), M.rgbToLab(b)));
  const script = `
import sys, json, numpy as np
from skimage.color import rgb2lab, deltaE_ciede2000
d = json.load(sys.stdin)
a = np.array([p[0] for p in d], dtype=float).reshape(-1,1,3) / 255.0
b = np.array([p[1] for p in d], dtype=float).reshape(-1,1,3) / 255.0
print(json.dumps(deltaE_ciede2000(rgb2lab(a), rgb2lab(b)).reshape(-1).tolist()))`;
  const theirs = JSON.parse(execFileSync(PY, ["-c", script], { input: JSON.stringify(pairs), encoding: "utf8", maxBuffer: 1 << 24 }));
  let maxAbs = 0, at = 0;
  for (let i = 0; i < mine.length; i++) { const d = Math.abs(mine[i] - theirs[i]); if (d > maxAbs) { maxAbs = d; at = i; } }
  const held = maxAbs < 1e-3;
  const m = `max |port - skimage| over 500 pairs = ${maxAbs.toExponential(4)} (limit 1e-3), worst pair ${JSON.stringify(pairs[at])} port=${f4(mine[at])} skimage=${f4(theirs[at])}`;
  held ? ok("C1 ", m) : no("C1 ", m);
  return held;
}

// C3 — identity. ref.png against itself is ΔE 0 and score 1.0 under EVERY rung and EVERY granularity.
// Its INPUT mutation offsets the candidate by one row: a region reader that silently mis-slices is
// invisible to every other check here.
function C3(M, half, mutate = false) {
  const A = M.decodePng(path.join(FIX, "ref.png"));
  const B = mutate ? shift(A, 0, 1) : clone(A);
  const grans = [["a whole", M.WHOLE(A.w, A.h)], ["b ybands", M.YBANDS(A.w, A.h)], ["c parts", M.PARTS()]];
  let worst = 0, where = "";
  for (const [g, regs] of grans) for (const s of regs.map((r) => statsOne(M, A, B, r)))
    for (const r of M.RUNGS) if (Math.abs(s[r.key]) > worst) { worst = Math.abs(s[r.key]); where = `${g} ${s.name} rung ${r.id}`; }
  const score = M.aggregate(grans[2][1].map((r) => statsOne(M, A, B, r)), "mean").score;
  const held = worst <= 1e-9 && Math.abs(score - 1) <= 1e-9;
  const m = `worst non-zero across 3 granularities x 6 rungs = ${worst.toExponential(3)}${where ? ` (${where})` : ""}, E4 score = ${score.toFixed(4)}`;
  held ? ok("C3 ", m) : no("C3 ", m);
  return held;
}

// C4a — a known swap measures EXACTLY, on synthetic input with no anti-aliasing.
// Paper #ffffff, a solid 20x20 ink block, and a 1px frame at #f8f8f8 (ΔE 1.40 from paper — BELOW the
// 2.3 mask threshold, ABOVE half of it). Recolour only the block; the ink-masked mean must equal that
// pair's ΔE to 1e-9. The source mutation halves the threshold: the frame joins the mask, its own ΔE is
// 0 because it was not recoloured, and the mean falls.
function C4a(M, half) {
  const A = blank(40, 40, HEX("#ffffff"));
  for (let x = 0; x < 40; x++) { setPx(A, x, 0, HEX("#f8f8f8")); setPx(A, x, 39, HEX("#f8f8f8")); setPx(A, 0, x, HEX("#f8f8f8")); setPx(A, 39, x, HEX("#f8f8f8")); }
  const B = clone(A);
  for (let y = 10; y < 30; y++) for (let x = 10; x < 30; x++) { setPx(A, x, y, HEX("#454545")); setPx(B, x, y, HEX("#0c3b2f")); }
  const s = statsOne(M, A, B, { name: "synthetic", x: 0, y: 0, w: 40, h: 40 });
  const want = M.deltaEHex("#454545", "#0c3b2f");
  const held = Math.abs(s.inkMean - want) <= 1e-9;
  const m = `inkMean = ${s.inkMean.toFixed(9)}, ΔE(#454545,#0c3b2f) = ${want.toFixed(9)}, |diff| = ${Math.abs(s.inkMean - want).toExponential(3)} (limit 1e-9); ink ${s.inkN}/${s.n} px, modal ${s.modalHex}`;
  held ? ok("C4a", m) : no("C4a", m);
  return held;
}

// C4b — the same swap on the REAL region, where anti-aliasing is in play. The plan asserted a BOUND,
// 0.5 x pairΔE <= inkMean <= pairΔE, on the assumption that at least half the ink mask is solid glyph
// core. MEASURED, the core is 35.1% of the mask, and the bound's lower leg fails — the assumption was
// never measured (plan error, AMENDMENTS E5). What the measurement gives instead is STRONGER than the
// bound the plan asked for: the non-core ink pixels are unchanged by an exact-match recolour, so their
// ΔE is exactly 0 and
//        inkMean  ===  (coreN / inkN) x pairΔE
// holds as an IDENTITY to 1e-9 on real anti-aliased input. The bound is replaced by that identity plus
// the plan's upper leg. This is not a loosened tolerance — it is a tighter assertion with the coverage
// term measured rather than guessed (the plan's own GOTCHA names that term and declines to predict it).
// Two mutations: a SOURCE one that dilutes the ink mean over the whole region breaks the identity; the
// plan's INPUT one (recolour half the core) leaves it holding at the new coverage, which is the
// positive control that the coverage term is real and not a fitted constant.
function C4b(M, half, mutate = false) {
  const A = M.decodePng(path.join(FIX, "m1-faithful.png"));
  const B = clone(A);
  const r = part(M, "title");
  const from = HEX("#454545"), to = HEX("#0c3b2f");
  let moved = 0, seen = 0;
  for (let y = r.y; y < r.y + r.h; y++) for (let x = r.x; x < r.x + r.w; x++) {
    const p = getPx(A, x, y);
    if (p[0] === from[0] && p[1] === from[1] && p[2] === from[2]) { seen++; if (!mutate || seen % 2 === 0) { setPx(B, x, y, to); moved++; } }
  }
  const s = statsOne(M, A, B, r);
  const pair = M.deltaEHex("#454545", "#0c3b2f");
  const coverage = moved / s.inkN;
  const want = coverage * pair;
  const identity = Math.abs(s.inkMean - want) <= 1e-9;
  const upper = s.inkMean <= pair;
  const held = identity && upper;
  const m = `inkMean = ${s.inkMean.toFixed(9)} === coverage ${(coverage * 100).toFixed(1)}% x ΔE ${pair.toFixed(4)} = ${want.toFixed(9)} (|diff| ${Math.abs(s.inkMean - want).toExponential(3)}, limit 1e-9)${identity ? "" : " IDENTITY BROKEN"}; <= pairΔE? ${upper ? "yes" : "NO"}; ${moved}/${seen} exact-#454545 px moved, ink mask ${s.inkN}/${s.n}`;
  held ? ok("C4b", m) : no("C4b", m);
  return held;
}

// C6 — metric agreement. Oklab ΔE (system/oklch.mjs:37, the repo's OWN colour math) must rank the
// same worst region as CIEDE2000. No mutation by design: a disagreement stops the run and is
// reported, it is not worked around.
async function C6(M, half) {
  const { hexToRgb, rgbToOklab } = await import(path.join(MAIN, "system", "oklch.mjs"));
  const oklabDE = (h1, h2) => {
    const a = rgbToOklab(hexToRgb(h1)), b = rgbToOklab(hexToRgb(h2));
    return Math.hypot(a.l - b.l, a.a - b.a, a.b - b.b);
  };
  const A = M.decodePng(path.join(FIX, "ref.png")), B = M.decodePng(path.join(FIX, "m1-wrong.png"));
  const stats = M.PARTS().map((r) => statsOne(M, A, B, r)).filter((s) => !s.tooSmall && !s.degenerate);
  const byCiede = [...stats].sort((a, b) => b.inkColourDeltaE - a.inkColourDeltaE)[0];
  const byOklab = [...stats].sort((a, b) => oklabDE(b.inkMeanHexA, b.inkMeanHexB) - oklabDE(a.inkMeanHexA, a.inkMeanHexB))[0];
  const held = byCiede.name === byOklab.name;
  const m = `worst region by CIEDE2000 = ${byCiede.name} (${f4(byCiede.inkColourDeltaE)}), by Oklab ΔE = ${byOklab.name} (${oklabDE(byOklab.inkMeanHexA, byOklab.inkMeanHexB).toFixed(4)}) — ${held ? "agree" : "DISAGREE: reported, not worked around"}`;
  held ? ok("C6 ", m) : no("C6 ", m);
  return held;
}

// C8 — THE HARNESS MEASURED WHAT IT CLAIMS TO. Under M1 the six dropped roles render at the same
// literal in both candidates, so avatar / chip / chip-text / chevron must be identical between
// m1-faithful.png and m1-wrong.png. This catches a wrong pack link, a stale PNG or a mis-parameterised
// harness — a different failure from anything C1–C7 reaches.
// The INPUT mutation substitutes m1-faithful for m1-wrong: C8 STAYS GREEN (the drops were never going
// to move) while the two mapped regions collapse to the floor and rung 1's signal vanishes. Reporting
// that is the point — C8 is a harness control, not a signal control, and it says so out loud.
function C8(M, half, mutate = false) {
  const ref = M.decodePng(path.join(FIX, "ref.png"));
  const fa = M.decodePng(path.join(FIX, "m1-faithful.png"));
  const wr = M.decodePng(path.join(FIX, mutate ? "m1-faithful.png" : "m1-wrong.png"));
  const drops = ["avatar", "chip", "chip-text", "chevron"];
  let worst = 0, where = "";
  for (const n of drops) {
    const r = part(M, n);
    const d = Math.abs(statsOne(M, ref, fa, r).mean - statsOne(M, ref, wr, r).mean);
    if (d > worst) { worst = d; where = n; }
  }
  const mapped = ["title", "subtitle"].map((n) => {
    const r = part(M, n);
    return `${n} floor=${f4(statsOne(M, ref, fa, r).mean)} signal=${f4(statsOne(M, ref, wr, r).mean)}`;
  }).join(", ");
  const held = worst <= 1e-9;
  const m = `drops worst |floor-signal| = ${worst.toExponential(3)}${where ? ` (${where})` : ""} (limit 1e-9); mapped: ${mapped}${mutate ? "  [m1-wrong SUBSTITUTED by m1-faithful]" : ""}`;
  held ? ok("C8 ", m) : no("C8 ", m);
  return held;
}

// C9 — RUNG 6 IS REGISTRATION-INVARIANT, which is the whole reason it is in the ladder. Shift the
// candidate by one pixel in x and one in y: inkColourDeltaE must move by < 0.5 while rung 5's
// inkMean moves by > 5. The source mutation makes rung 6 compare per-pixel ink instead of the two ink
// AVERAGES; the one-pixel shift then moves it like rung 5 and the invariance is gone.
function C9(M, half) {

  const held = d6 < 0.5 && d5 > 5;
  const m = `1px shift: rung 6 ${f4(s0.inkColourDeltaE)} -> ${f4(s1.inkColourDeltaE)} (Δ ${f4(d6)}, limit <0.5); rung 5 ${f4(s0.inkMean)} -> ${f4(s1.inkMean)} (Δ ${f4(d5)}, limit >5)`;

  return held;

// C10 — RUNG 6'S OWN LIMITATION IS STATED, NOT HIDDEN. Under M2 the WRONG pack moves the surface role
// #f8f8f8 -> #ffffff. Rung 6 then penalises regions whose INK NEVER CHANGED — all four of M1's
// dropped-role regions — because their anti-aliased edge pixels blend the unchanged ink against a
// MOVED paper, so the ink AVERAGE moves. Assert that penalty is real and report all four regions.
//
// The plan named a different mechanism and a different mutation: "each image's modal colour moves",
// mutated by holding the modal fixed to the reference's. Both were DRIVEN and the plan is wrong on
// this fixture (AMENDMENTS E6). The modal does not move on three of the four regions — the avatar's
// mode is the avatar circle, the chip's is the chip fill — so holding it fixed is a semantic no-op
// and C10 stayed green under it (measured). Where the modal DOES move (chevron: #f8f8f8 -> #ffffff)
// the penalty still survives the mutation, because the mechanism is the blend, not the mode.
// The mutation that does hide the limitation is a CORE-ONLY ink mask (threshold 2.3 -> 20, excluding
// the anti-aliased edge): the penalty collapses to 0.0000 on avatar and chevron and goes NEGATIVE on
// the chip, i.e. M2 reads BETTER than M1. That is the "fix" a reader would plausibly reach for, and
// it is the one that must be shown to destroy the control.
function C10(M, half) {
  const ref = M.decodePng(path.join(FIX, "ref.png"));
  const w1 = M.decodePng(path.join(FIX, "m1-wrong.png")), w2 = M.decodePng(path.join(FIX, "m2-wrong.png"));
  const rows = ["avatar", "chip", "chip-text", "chevron"].map((n) => {   // M1's dropped-ink regions
    const r = part(M, n);
    const a = statsOne(M, ref, w1, r), b = statsOne(M, ref, w2, r);
    return { n, m1: a.inkColourDeltaE, m2: b.inkColourDeltaE, d: b.inkColourDeltaE - a.inkColourDeltaE,
             modA: b.modalHex, modB: b.modalHexB };
  });
  const worst = rows.reduce((x, y) => (y.d > x.d ? y : x));
  const held = worst.d > 0.5;
  const m = `ink-unchanged regions, rung 6 M1 -> M2: ` +
    rows.map((x) => `${x.n} ${f4(x.m1)}->${f4(x.m2)} (${x.d >= 0 ? "+" : ""}${f4(x.d)}${x.modA !== x.modB ? ", modal MOVED " + x.modA + "->" + x.modB : ""})`).join("; ") +
    `  worst penalty ${f4(worst.d)} on ${worst.n} (limit >0.5)`;
  held ? ok("C10", m) : no("C10", m);
  return held;
}

async function mutated(id, edits) {
  const src = readFileSync(SRC, "utf8");
  let out = src, applied = 0;
  for (const [from, to] of edits) {
    if (!out.includes(from)) { console.log(`  ${id} MUTATION DID NOT APPLY — source fragment not found`); return null; }
    out = out.replace(from, to); applied++;
  }
  if (out === src) { console.log(`  ${id} MUTATION DID NOT APPLY — source unchanged after ${applied} replacement(s)`); return null; }
  const p = path.join(HERE, `compare.mut.${id}.mjs`);
  writeFileSync(p, out);
  return await import(p + `?v=${Date.now()}`);
}
